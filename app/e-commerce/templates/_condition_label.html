{# Condition label macro: «نو» or «دست دوم» + cleanliness stars. Call: {{ condition_label(condition, stars, sells_used) }} - condition: 'new' or 'used' (product.condition) - stars: 1–5 when used, None otherwise (product.condition_stars) - sells_used: site.settings.sells_used — a site that sells new only hides the whole flow, so the macro renders nothing at all. Stars: 5 = like new, 1 = a wreck. #} {% macro condition_label(condition, stars, sells_used) -%} {% if sells_used %} {% if condition == 'used' %} دست دوم {% if stars %} {%- for i in range(1, 6) -%}{{ '★' if i <= stars else '☆' }}{%- endfor -%} {% endif %} {% else %} نو {% endif %} {% endif %} {%- endmacro %}
